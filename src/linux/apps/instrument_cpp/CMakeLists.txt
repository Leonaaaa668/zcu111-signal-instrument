cmake_minimum_required(VERSION 3.16)
project(zcu111_instrument VERSION 0.10.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(INSTRUMENT_WARNINGS_AS_ERRORS "Fail compilation on warnings" OFF)
if(INSTRUMENT_WARNINGS_AS_ERRORS)
    add_compile_options(-Werror)
endif()
option(INSTRUMENT_BUILD_GUI "Build the standalone Qt front panel" ON)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
add_library(instrument_core STATIC src/core.cpp src/waveform.cpp src/clock.cpp src/storage.cpp src/tx.cpp src/rx_power.cpp src/tuning.cpp src/sweep.cpp src/tx_power.cpp src/spectrum_tools.cpp)
target_include_directories(instrument_core PUBLIC src include)
target_link_libraries(instrument_core PUBLIC Threads::Threads)
target_compile_options(instrument_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(zcu111-measure src/cli.cpp)
target_link_libraries(zcu111-measure PRIVATE instrument_core)
add_executable(zcu111-stream src/stream_cli.cpp)
add_executable(zcu111-sweep src/sweep_cli.cpp)
target_link_libraries(zcu111-sweep PRIVATE instrument_core)
target_link_libraries(zcu111-stream PRIVATE instrument_core Threads::Threads)
add_executable(zcu111-wavegen src/wave_cli.cpp)
target_link_libraries(zcu111-wavegen PRIVATE instrument_core)
add_executable(zcu111-clock src/clock_cli.cpp)
target_link_libraries(zcu111-clock PRIVATE instrument_core Threads::Threads)
add_executable(zcu111-tx src/tx_cli.cpp)
target_link_libraries(zcu111-tx PRIVATE instrument_core Threads::Threads)
install(TARGETS zcu111-measure zcu111-stream zcu111-sweep zcu111-wavegen zcu111-clock zcu111-tx RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS deploy/check-target DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME zcu111-instrument-check)
if(INSTRUMENT_BUILD_GUI)
    find_package(QT NAMES Qt5 Qt6 REQUIRED COMPONENTS Core Gui Widgets)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Gui Widgets)
    add_executable(zcu111-instrument src/gui.cpp)
    target_link_libraries(zcu111-instrument PRIVATE instrument_core Threads::Threads
        Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Gui Qt${QT_VERSION_MAJOR}::Widgets)
    target_compile_options(zcu111-instrument PRIVATE -Wall -Wextra -Wpedantic)
    install(TARGETS zcu111-instrument RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(DIRECTORY resources/ DESTINATION ${CMAKE_INSTALL_DATADIR}/zcu111-instrument)
set(INSTRUMENT_SYSTEMD_DIR "${CMAKE_INSTALL_LIBDIR}/systemd/system" CACHE STRING "systemd system unit directory")
set(INSTRUMENT_UDEV_DIR "${CMAKE_INSTALL_LIBDIR}/udev/rules.d" CACHE STRING "udev rules directory")
if(INSTRUMENT_BUILD_GUI)
    install(PROGRAMS deploy/launch-wayland DESTINATION ${CMAKE_INSTALL_BINDIR}
        RENAME zcu111-instrument-launch)
    install(FILES deploy/zcu111-instrument.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES deploy/zcu111-instrument.service DESTINATION ${INSTRUMENT_SYSTEMD_DIR})
endif()
install(FILES deploy/90-zcu111-spectrum.rules DESTINATION ${INSTRUMENT_UDEV_DIR})
include(CTest)

    add_executable(instrument-rx-power-tests tests/rx_power_tests.cpp)
    target_link_libraries(instrument-rx-power-tests PRIVATE instrument_core)
    add_test(NAME rx-power-calibration COMMAND instrument-rx-power-tests)

    target_link_libraries(instrument-core-tests PRIVATE instrument_core)
    add_test(NAME core-contract COMMAND instrument-core-tests
        ${CMAKE_CURRENT_SOURCE_DIR}/resources ${CMAKE_CURRENT_BINARY_DIR}/test-evidence)
    add_executable(instrument-waveform-tests tests/waveform_tests.cpp)
    target_link_libraries(instrument-waveform-tests PRIVATE instrument_core)
    add_test(NAME waveform-contract COMMAND instrument-waveform-tests
        ${CMAKE_CURRENT_BINARY_DIR}/waveform-test-evidence)

    target_link_libraries(instrument-control-tests PRIVATE instrument_core Threads::Threads)
    add_test(NAME control-clock-storage COMMAND instrument-control-tests
        ${CMAKE_CURRENT_SOURCE_DIR}/resources ${CMAKE_CURRENT_BINARY_DIR}/control-test-evidence)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
